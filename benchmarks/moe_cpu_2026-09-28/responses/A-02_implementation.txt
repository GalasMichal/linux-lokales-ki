```typescript
function clampLonLat({ lon, lat }: { lon: number; lat: number }): { lon: number; lat: number } {
  if (!Number.isFinite(lon) || !Number.isFinite(lat)) {
    throw new TypeError('invalid lon/lat');
  }
  return {
    lon: Math.max(-180, Math.min(180, lon)),
    lat: Math.max(-90, Math.min(90, lat))
  };
}
```
