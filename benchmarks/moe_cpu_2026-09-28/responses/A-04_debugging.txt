```typescript
export function mergeUniqueIds(a: string[], b: string[]): string[] {
  const out: string[] = [];
  for (let i = 0; i < a.length; i++) {
    if (!out.includes(a[i])) out.push(a[i]);
  }
  for (const id of b) {
    if (out.includes(id)) continue;
    out.push(id);
  }
  return out.filter(Boolean);
}
```

Die Hauptprobleme waren:
- Der erste Loop ging über `a.length` statt `a.length - 1`
- `indexOf(id) > 0` sollte `includes(id)` sein, um Duplikate korrekt zu erk
