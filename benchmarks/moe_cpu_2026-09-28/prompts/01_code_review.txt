```typescript
import Map from 'ol/Map';
import View from 'ol/View';
import TileLayer from 'ol/layer/Tile';
import OSM from 'ol/source/OSM';
import VectorLayer from 'ol/layer/Vector';
import VectorSource from 'ol/source/Vector';
import { fromLonLat } from 'ol/proj';
import Feature from 'ol/Feature';
import Point from 'ol/geom/Point';

export function createMap(target: string, lon: number, lat: number) {
  const source = new VectorSource();
  const layer = new VectorLayer({ source });
  const map = new Map({
    target,
    layers: [new TileLayer({ source: new OSM() }), layer],
    view: new View({
      center: fromLonLat([lon, lat]),
      zoom: 12,
    }),
  });

  // BUG-ish patterns for review:
  setInterval(() => {
    fetch('/api/positions')
      .then((r) => r.json())
      .then((data) => {
        source.clear();
        data.forEach((p: any) => {
          source.addFeature(
            new Feature({
              geometry: new Point(fromLonLat([p.lon, p.lat])),
              name: p.name,
            }),
          );
        });
        map.getView().setCenter(fromLonLat([data[0].lon, data[0].lat]));
      });
  }, 1000);

  document.getElementById(target)!.style.height = '100vh';
  return map;
}
```
